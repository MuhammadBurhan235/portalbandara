import { useState, useEffect } from 'react';
import { Star, Wifi } from 'lucide-react';
import { apiClient } from './api/axios';

function App() {
  const [rating, setRating] = useState<number>(0);
  const [hoveredRating, setHoveredRating] = useState<number>(0);
  const [kategori, setKategori] = useState<string>('');
  const [komentar, setKomentar] = useState<string>('');
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [lokasi, setLokasi] = useState<string>('Portal Utama');
  
  // 1. Tambahkan State untuk menampung data dari Mikrotik
  const [linkLogin, setLinkLogin] = useState<string>('');
  const [macAddress, setMacAddress] = useState<string>('');

  useEffect(() => {
    const queryParams = new URLSearchParams(window.location.search);
    
    // Tangkap Lokasi
    const lokasiDariURL = queryParams.get('loc');
    if (lokasiDariURL) setLokasi(lokasiDariURL.replace(/_/g, ' '));

    // 2. Tangkap URL Login Mikrotik dan MAC Address
    const urlLoginMikrotik = queryParams.get('link-login');
    const mac = queryParams.get('mac');
    
    if (urlLoginMikrotik) setLinkLogin(urlLoginMikrotik);
    if (mac) setMacAddress(mac);
  }, []);

  // 3. Ubah logika pemberian akses internet
  const prosesKoneksiWiFi = () => {
    if (linkLogin) {
      // Jika terdeteksi URL Mikrotik, lempar kembali dengan kredensial
      // CATATAN: 'guest' dan password '123' adalah contoh. 
      // Anda harus menyesuaikan ini dengan User Hotspot yang dibuat di Router Mikrotik Bandara.
      const urlTujuan = `${linkLogin}?username=guest&password=123`;
      window.location.href = urlTujuan;
    } else {
      // Jika link-login kosong (misal Anda sedang testing di laptop/lokal)
      console.log(`MAC Address terdeteksi: ${macAddress || 'Tidak ada'}`);
      window.location.href = 'https://google.com';
    }
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setIsLoading(true);

    try {
      await apiClient.post('/feedback', {
        lokasi_scan: lokasi,
        rating: rating,
        kategori_layanan: kategori,
        komentar: komentar,
        // Opsional: Anda bisa mengirim macAddress ke backend di sini jika nanti database Laravel diupdate
      });
      prosesKoneksiWiFi();
    } catch (error) {
      console.error(error);
      alert('Gagal menyimpan feedback, tapi Anda tetap akan dihubungkan ke Wi-Fi.');
      prosesKoneksiWiFi();
    } finally {
      setIsLoading(false);
    }
  };

  const handleSkip = async () => {
    setIsLoading(true);
    try {
      await apiClient.post('/feedback', {
        lokasi_scan: lokasi,
        rating: 0,
        kategori_layanan: 'Skipped',
        komentar: ''
      });
    } catch (error) {
      console.error(error);
    } finally {
      prosesKoneksiWiFi();
    }
  };
