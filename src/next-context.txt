import { useState } from 'react';
import { Star, Plane } from 'lucide-react';
import { apiClient } from './api/axios'; // <-- Tambahkan import ini

function App() {
  const [rating, setRating] = useState<number>(0);
  const [hoveredRating, setHoveredRating] = useState<number>(0);
  const [kategori, setKategori] = useState<string>('');
  const [komentar, setKomentar] = useState<string>('');
  const [isLoading, setIsLoading] = useState<boolean>(false); // <-- Tambah state loading

  const listKategori = ['Kebersihan', 'Kecepatan Layanan', 'Fasilitas', 'Keamanan', 'Lainnya'];

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setIsLoading(true);

    try {
      // Mengirim POST request ke endpoint /feedback di Laravel
      await apiClient.post('/feedback', {
        lokasi_scan: 'Portal Utama', // Nanti ini bisa dibuat dinamis
        rating: rating,
        kategori_layanan: kategori,
        komentar: komentar
      });

      alert('Terima kasih atas penilaian Anda! Mengalihkan ke internet...');
      
      // Reset form setelah berhasil
      setRating(0);
      setKategori('');
      setKomentar('');
      
      // Di sini nanti logika untuk tembak API Mikrotik untuk buka Wi-Fi
      
    } catch (error) {
      console.error(error);
      alert('Terjadi kesalahan saat mengirim data. Silakan coba lagi.');
    } finally {
      setIsLoading(false);
    }
  };

  // ... (kode bagian return UI biarkan sama persis seperti sebelumnya)
